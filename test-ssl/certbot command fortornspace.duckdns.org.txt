sudo python3 -m venv /opt/certbot/
sudo /opt/certbot/bin/pip install --upgrade pip
sudo /opt/certbot/bin/pip install certbot
sudo ln -s /opt/certbot/bin/certbot /usr/local/bin/certbot
sudo /opt/certbot/bin/pip install certbot_dns_duckdns
sudo /opt/certbot/bin/pip install certbot_dns_duckdns -U
sudo /opt/certbot/bin/certbot --version
sudo /opt/certbot/bin/certbot certonly \
  --non-interactive \
  --agree-tos \
  --email andromedavirgoa@gmail.com \
  --preferred-challenges dns \
  --authenticator dns-duckdns \
  --dns-duckdns-token <INSERT_YOUR_DUCKDNS_TOKEN_HERE> \
  --dns-duckdns-propagation-seconds 60 \
  -d "tornspace.duckdns.org"
